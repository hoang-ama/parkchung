server {
    server_name parkchung.com www.parkchung.com;

    root /var/www/parkchung/client/customer;
    #root /var/www/parkchung/client;
    index index.html;

    location /api/ {
        proxy_pass http://127.0.0.1:3001/api/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location ^~ /host/ {
        alias /var/www/parkchung/client/host/dist/;
        try_files $uri $uri/ /host/index.html;
    }

    location ^~ /admin/ {
        alias /var/www/parkchung/client/admin/html/;
        index login.html;
    }
# Admin static
    location ^~ /admin/css/ {
        alias /var/www/parkchung/client/admin/css/; try_files $uri =404;
    }
    location ^~ /admin/js/  {
        alias /var/www/parkchung/client/admin/js/;  try_files $uri =404;
    }
# config.js dùng chung/Customer
    location ^~ /customer/js/ {
        alias /var/www/parkchung/client/customer/js/; try_files $uri =404;
    }
    location ^~ /customer/css/ {
        alias /var/www/parkchung/client/customer/css/; try_files $uri =404;
    }
# Assets
    location ^~ /assets/ {
        alias /var/www/parkchung/client/assets/;
		access_log off;
        expires 30d;
    }
# Uploads for Vite
    location /uploads/ {
    alias /var/www/parkchung/server/public/uploads/;
    expires 1y;
    add_header Cache-Control "public, immutable";
    }

    location ~* \.(png|jpe?g|gif|webp|svg)$ {
    expires 7d;
    add_header Cache-Control "public, max-age=604800";
    }
    location / {
        try_files $uri $uri/ =404;
    }
    listen 443 ssl; # managed by Certbot
    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/parkchung.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/parkchung.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
}

server {
    if ($host = www.parkchung.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    if ($host = parkchung.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    listen [::]:80;
    server_name parkchung.com www.parkchung.com;
    return 404; # managed by Certbot
}
